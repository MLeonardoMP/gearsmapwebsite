================================================================================
INSTALLATION METHODS
================================================================================

METHOD 1 - INTERACTIVE (RECOMMENDED):
git clone https://github.com/VoltAgent/awesome-claude-code-subagents.git
cd awesome-claude-code-subagents
./install-agents.sh

METHOD 2 - INSTALL SPECIFIC PLUGIN:
claude plugin install voltagent-core-dev

METHOD 3 - STANDALONE INSTALLER:
curl -sO https://raw.githubusercontent.com/VoltAgent/awesome-claude-code-subagents/main/install-agents.sh
chmod +x install-agents.sh
./install-agents.sh

METHOD 4 - MANUAL AGENT INSTALLATION:
mkdir -p ~/.claude/agents
git clone https://github.com/VoltAgent/awesome-claude-code-subagents.git
cp awesome-claude-code-subagents/categories/01-core-development/frontend-developer.md ~/.claude/agents/

METHOD 5 - SLASH COMMANDS:
git clone https://github.com/hesreallyhim/awesome-claude-code.git
cp -r awesome-claude-code/resources/slash-commands/* ~/.claude/commands/

METHOD 6 - HOOKS:
mkdir -p ~/.claude/hooks
git clone https://github.com/bartolli/claude-code-typescript-hooks.git
cp -r claude-code-typescript-hooks/* ~/.claude/hooks/
